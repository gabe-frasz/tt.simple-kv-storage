# Test Cases - In-Memory Key-Value Database
# Format: Commands followed by the corresponding expected response

# ---------------------------------------------------------------------------
# 1. Basic Commands (Keys without extensions)
# ---------------------------------------------------------------------------
ADD user_1 Alice
# Expected: OK
GET user_1
# Expected: Alice

ADD welcome_msg Hello Database World
# Expected: OK
GET welcome_msg
# Expected: Hello Database World

# Overwrite value on the same key
ADD user_1 Robert
# Expected: OK
GET user_1
# Expected: Robert

# Nonexistent key
GET nonexistent_key
# Expected: ERROR: key not found

# ---------------------------------------------------------------------------
# 2. Extension: CPF Validator (cpf.lua)
# ---------------------------------------------------------------------------
# Valid insertion and formatting on GET (000.000.000-00)
ADD cpf_joe 12345678909
# Expected: OK
GET cpf_joe
# Expected: 123.456.789-09

ADD cpf_maria 52998224725
# Expected: OK
GET cpf_maria
# Expected: 529.982.247-25

# Uniqueness Rule: Rejects same CPF on a different key
ADD cpf_other_joe 12345678909
# Expected: ERROR: CPF already registered under key cpf_joe

# Uniqueness Rule: Allows rewriting the same CPF on the same key
ADD cpf_joe 12345678909
# Expected: OK
GET cpf_joe
# Expected: 123.456.789-09

# Rejection of CPFs with repeated digits (even if they pass mod 11)
ADD cpf_rep1 11111111111
# Expected: ERROR: Invalid CPF: repeated digits

ADD cpf_rep0 00000000000
# Expected: ERROR: Invalid CPF: repeated digits

ADD cpf_rep9 99999999999
# Expected: ERROR: Invalid CPF: repeated digits

# Rejection of invalid check digits
ADD cpf_invalid 12345678900
# Expected: ERROR: Invalid CPF check digit

# Rejection of CPF with punctuation on input (input must be strict: digits only)
ADD cpf_with_punctuation 123.456.789-09
# Expected: ERROR: CPF must contain exactly 11 numeric digits without formatting

# Rejection of wrong length or invalid characters
ADD cpf_short 1234567890
# Expected: ERROR: CPF must contain exactly 11 numeric digits without formatting

ADD cpf_long 123456789012
# Expected: ERROR: CPF must contain exactly 11 numeric digits without formatting

ADD cpf_letters 1234567890a
# Expected: ERROR: CPF must contain exactly 11 numeric digits without formatting

# ---------------------------------------------------------------------------
# 3. Extension: Date Formatter (data.lua)
# ---------------------------------------------------------------------------
# Valid insertion and formatting on GET (DD/MM/YYYY)
ADD data_birth_joe 2000-01-23
# Expected: OK
GET data_birth_joe
# Expected: 23/01/2000

ADD data_event 2024-12-31
# Expected: OK
GET data_event
# Expected: 31/12/2024

# Regular leap year (divisible by 4)
ADD data_leap_2024 2024-02-29
# Expected: OK
GET data_leap_2024
# Expected: 29/02/2024

# Century leap year (divisible by 400)
ADD data_leap_2000 2000-02-29
# Expected: OK
GET data_leap_2000
# Expected: 29/02/2000

# Non-leap century year (divisible by 100 but not by 400) -> Rejects
ADD data_non_leap_1900 1900-02-29
# Expected: ERROR: invalid day for month

ADD data_non_leap_2100 2100-02-29
# Expected: ERROR: invalid day for month

# Common year (non-leap) -> Rejects Feb 29
ADD data_common_2023 2023-02-29
# Expected: ERROR: invalid day for month

# Out-of-range days per month: April has 30 days -> Rejects 31
ADD data_april_31 2023-04-31
# Expected: ERROR: invalid day for month

# Out-of-range month (13 and 00)
ADD data_month_13 2023-13-01
# Expected: ERROR: invalid month

ADD data_month_0 2023-00-10
# Expected: ERROR: invalid month

# Day 00 or day above 31
ADD data_day_0 2023-05-00
# Expected: ERROR: invalid day for month

ADD data_day_32 2023-01-32
# Expected: ERROR: invalid day for month

# Loose formats without leading zeros -> Rejects
ADD data_loose1 2023-1-5
# Expected: ERROR: date must be in YYYY-MM-DD format

ADD data_loose2 23-01-2000
# Expected: ERROR: date must be in YYYY-MM-DD format

# ---------------------------------------------------------------------------
# 4. Parser Robustness and Syntax Error Handling
# ---------------------------------------------------------------------------
# Unknown command
INVALID_COMMAND key value
# Expected: ERROR: unknown command: INVALID_COMMAND

# Incomplete commands (missing arguments)
ADD
# Expected: ERROR: key not provided

ADD key_without_value
# Expected: ERROR: value not provided

GET
# Expected: ERROR: key not provided

# Excess arguments
GET key1 extra_argument
# Expected: ERROR: too many arguments

EXIT extra_argument
# Expected: ERROR: too many arguments

# Program termination
EXIT
