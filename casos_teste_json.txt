# Test Cases - Free-Theme Proposed Extension: JSON (json.lua)
# Prefix: json_
# Format: Commands followed by the corresponding expected response

# ---------------------------------------------------------------------------
# 1. Full Insertion and Table Formatting on GET
# ---------------------------------------------------------------------------
# Flat object with string, number (integer) and boolean types
ADD json_user {"name": "Alice", "age": 30, "active": true}
# Expected: OK
GET json_user
# Expected: +--------+---------+
# Expected: | FIELD  | VALUE   |
# Expected: +--------+---------+
# Expected: | active | true    |
# Expected: | age    | 30      |
# Expected: | name   | "Alice" |
# Expected: +--------+---------+

# Flat object with decimal and negative numbers
ADD json_metric {"count": 100, "temp": -5.5, "rate": 0.75}
# Expected: OK
GET json_metric
# Expected: +-------+-------+
# Expected: | FIELD | VALUE |
# Expected: +-------+-------+
# Expected: | count | 100   |
# Expected: | rate  | 0.75  |
# Expected: | temp  | -5.5  |
# Expected: +-------+-------+

# Empty JSON object
ADD json_empty {}
# Expected: OK
GET json_empty
# Expected: {}

# ---------------------------------------------------------------------------
# 2. Atomic Idempotent Partial Update (Patch) via Lookup on ADD
# ---------------------------------------------------------------------------
# Update existing field (@field=value)
ADD json_user @age=31
# Expected: OK
GET json_user
# Expected: +--------+---------+
# Expected: | FIELD  | VALUE   |
# Expected: +--------+---------+
# Expected: | active | true    |
# Expected: | age    | 31      |
# Expected: | name   | "Alice" |
# Expected: +--------+---------+

# Add new field with string containing spaces (+@field=value)
ADD json_user +@city="Sao Paulo"
# Expected: OK
GET json_user
# Expected: +--------+-------------+
# Expected: | FIELD  | VALUE       |
# Expected: +--------+-------------+
# Expected: | active | true        |
# Expected: | age    | 31          |
# Expected: | city   | "Sao Paulo" |
# Expected: | name   | "Alice"     |
# Expected: +--------+-------------+

# Change boolean field
ADD json_user @active=false
# Expected: OK
GET json_user
# Expected: +--------+-------------+
# Expected: | FIELD  | VALUE       |
# Expected: +--------+-------------+
# Expected: | active | false       |
# Expected: | age    | 31          |
# Expected: | city   | "Sao Paulo" |
# Expected: | name   | "Alice"     |
# Expected: +--------+-------------+

# Remove existing field (-@field)
ADD json_user -@age
# Expected: OK
GET json_user
# Expected: +--------+-------------+
# Expected: | FIELD  | VALUE       |
# Expected: +--------+-------------+
# Expected: | active | false       |
# Expected: | city   | "Sao Paulo" |
# Expected: | name   | "Alice"     |
# Expected: +--------+-------------+

# Idempotency on removal: removing an already missing field does not error
ADD json_user -@age
# Expected: OK
GET json_user
# Expected: +--------+-------------+
# Expected: | FIELD  | VALUE       |
# Expected: +--------+-------------+
# Expected: | active | false       |
# Expected: | city   | "Sao Paulo" |
# Expected: | name   | "Alice"     |
# Expected: +--------+-------------+

# Idempotency on removal: removing a field that never existed does not error
ADD json_user -@nonexistent_field
# Expected: OK
GET json_user
# Expected: +--------+-------------+
# Expected: | FIELD  | VALUE       |
# Expected: +--------+-------------+
# Expected: | active | false       |
# Expected: | city   | "Sao Paulo" |
# Expected: | name   | "Alice"     |
# Expected: +--------+-------------+

# Idempotency on addition: adding an existing field with the same value does not error
ADD json_user +@city="Sao Paulo"
# Expected: OK
GET json_user
# Expected: +--------+-------------+
# Expected: | FIELD  | VALUE       |
# Expected: +--------+-------------+
# Expected: | active | false       |
# Expected: | city   | "Sao Paulo" |
# Expected: | name   | "Alice"     |
# Expected: +--------+-------------+

# Idempotency on new key: patch creates the document directly
ADD json_new +@status="active"
# Expected: OK
GET json_new
# Expected: +--------+----------+
# Expected: | FIELD  | VALUE    |
# Expected: +--------+----------+
# Expected: | status | "active" |
# Expected: +--------+----------+

# ---------------------------------------------------------------------------
# 3. Full JSON Document Overwrite
# ---------------------------------------------------------------------------
ADD json_user {"name": "Bob", "role": "Admin"}
# Expected: OK
GET json_user
# Expected: +-------+---------+
# Expected: | FIELD | VALUE   |
# Expected: +-------+---------+
# Expected: | name  | "Bob"   |
# Expected: | role  | "Admin" |
# Expected: +-------+---------+

# ---------------------------------------------------------------------------
# 4. Invalid Inputs and Rejections on ADD
# ---------------------------------------------------------------------------
# Key without double quotes
ADD json_bad1 {name: "Alice"}
# Expected: ERROR: JSON key must be enclosed in double quotes

# Wrong delimiters (does not open or close with braces)
ADD json_bad2 "name": "Alice"
# Expected: ERROR: value must be a JSON object '{...}' or an update command (@field=value, +@field=value, -@field)

# Invalid value type (unquoted and not numeric/boolean)
ADD json_bad3 {"age": unquoted_value}
# Expected: ERROR: invalid or unsupported value type (allowed: quoted string, number or boolean)

# Unterminated string
ADD json_bad4 {"name": "Alice}
# Expected: ERROR: unterminated string

# Invalid update syntax
ADD json_user @
# Expected: ERROR: invalid update syntax (use @field=value, +@field=value or -@field)

# Invalid value in patch
ADD json_user @age=unquoted_value
# Expected: ERROR: invalid value in patch: invalid or unsupported value type (allowed: quoted string, number or boolean)

# Nonexistent key on GET
GET json_not_registered
# Expected: ERROR: key not found

# Program termination
EXIT
